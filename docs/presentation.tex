\section{Introduction \& Motivation}
% =============================================================================

\begin{frame}{Base Paper: Findings \& Drawbacks (He et al., 2025)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Key Findings of Base Paper}}
        \begin{itemize}\itemsep=2pt\footnotesize
          \item \textbf{The ``Valley of Code Reasoning''}: Intermediate data scaling exhibits a sharp performance dip where student reasoning degrades before recovering.
          \item \textbf{Entangled Token Generation}: Jointly predicting rationale and code forces the model to resolve abstract math and low-level syntax simultaneously.
          \item \textbf{CoT Fragility in Code}: Standard Chain-of-Thought prompts frequently trigger premature convergence to suboptimal brute-force loops ($O(N^2)$).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Critical Drawbacks of Base Paper}}
        \begin{itemize}\itemsep=2pt\footnotesize
          \item \textbf{No Decoupled Planning}: Lacks an explicit System-2 stage to identify the algorithmic paradigm (DP, Graphs, Greedy) before code synthesis.
          \item \textbf{Unfiltered Teacher Contamination}: Assumes teacher rationales are always sound; hallucinated or mismatched traces pollute student supervision.
          \item \textbf{High Compute Dependency}: Evaluated only on multi-billion parameter models ($\ge 7\text{B}$), lacking accessible QLoRA solutions for sub-1B models on consumer GPUs.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{3pt}
  \begin{exampleblock}{Our Approach to Address These Drawbacks}
    \footnotesize
    \textbf{1. Decoupled Algorithmic Conditioning} (Stage 1 Classifier) + \textbf{2. Semantic Rationale Filter} (purging noise) + \textbf{3. Stage 2 Conditioned Synthesis} + \textbf{4. Consumer-Grade QLoRA} on single 15GB T4 GPU.
  \end{exampleblock}
\end{frame}

\section{Problem Statement \& Objectives}
% =============================================================================

\begin{frame}{Problem Statement}
  \begin{alertblock}{Core Vulnerabilities in Existing Code Distillation}
    \begin{enumerate}\itemsep=6pt
      \item \textbf{Unstructured Monolithic CoT Generation}: Student models attempt to predict algorithmic intuition and raw syntax simultaneously, leading to hallucinated loops and asymptotic timeout errors ($O(N^2)$ instead of $O(N \log N)$).
      \item \textbf{Teacher Trace Contamination}: Commercial teacher LLMs generate deceptive or mislabeled rationales that diverge from ground-truth paradigms. Distilling noisy traces degrades student reasoning.
      \item \textbf{High Compute Barriers}: Full parameter fine-tuning of multi-billion parameter models is inaccessible to standard edge and consumer hardware.
    \end{enumerate}
  \end{alertblock}
\end{frame}

          \item \textbf{Goal}: Eliminates search disorientation by anchoring on the committed paradigm before token synthesis.

          \item \textbf{Input Condition}: $\mathcal{I}_{\text{base}} = [\text{PROBLEM}]$.
          \item \textbf{Prompt Format}: Direct problem statement without \texttt{[ALGORITHM CATEGORY]} token.
          \item \textbf{Supervised Target}: $[\text{RATIONALE}] + [\text{CODE}]$.

        \end{itemize}

  \begin{alertblock}{\textbf{Mathematical Formulation of Decoupled Generation}}
    \scriptsize
    $$\log P(\text{code}, \text{rationale} \mid \text{problem}) \approx \sum_{\text{tag}} \underbrace{P(\text{tag} \mid \text{problem})}_{\text{Stage 1 Classifier}} \cdot \underbrace{P(\text{code}, \text{rationale} \mid \text{tag}, \text{problem})}_{\text{Stage 2 Conditioned Generator}}$$
  \end{alertblock}
\end{frame}

\begin{frame}{Diagnostic Insights: Few-Shot Dynamics \& Linear Head Penalty}
  \begin{alertblock}{Scientific Root-Cause Analysis}
    \begin{enumerate}\itemsep=3pt\footnotesize
      \item \textbf{Severe Few-Shot Sample Sparsity}:
        \begin{itemize}\scriptsize
          \item Training on 64 clean problems across 10 categories yields $\approx 6$ problems per class.
          \item Loss falling rapidly ($5.38 \to 0.77$) demonstrates swift adapter adaptation on the few-shot split, while generalization on unseen problems requires dataset scaling.
        \end{itemize}
      \item \textbf{The Random Linear Head Penalty on Causal LMs}:
        \begin{itemize}\scriptsize
          \item \texttt{Qwen2.5} is an autoregressive generative model. Newly initialized classification projections (\texttt{score.weight}) must align categorical boundaries from scratch under low sample counts.
        \end{itemize}
      \item \textbf{Syntax Guarding \& AST Compilation in Sandbox}:
        \begin{itemize}\scriptsize
          \item Free-form generation during early SFT epochs occasionally leaks markdown formatting; the sandbox AST compiler isolates errors before test execution.
        \end{itemize}
      \item \textbf{Direct Motivation for Empirical Scaling Sweep}:
        \begin{itemize}\scriptsize
          \item Directly validates the necessity of our class-balanced partitioner ($10\%, 25\%, 50\%, 100\%$) to track the transition from few-shot adaptation to generalizable reasoning.
        \end{itemize}
    \end{enumerate}
  \end{alertblock}
